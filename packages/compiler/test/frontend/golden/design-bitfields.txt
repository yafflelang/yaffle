module main.yfl (main)
enum MemKind (MemKind) : u8 { Sys = 0, Gpu = 1 }
root B = B
struct B [B] (endian endian: endian = le)  {usesEndian}
  bits u8 lsb {
    compressed: bool @0:1
    kind: enum MemKind @1:3
    _: u8 @4:4  input absent
  }
  bits u16 lsb {
    sizeHigh: u16 @0:12
    level: u16 @12:4
  }
  bits u16 msb {
    hi: u8 @12:4
    lo: u8 @8:4
    _: u16 @0:8  input absent
  }
  bits i8 lsb {
    s: i8 @0:3
    t: i8 @3:5
  }
