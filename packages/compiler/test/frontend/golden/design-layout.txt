module main.yfl (main)
root Chunk = Chunk
struct Chunk [Chunk] (endian endian: endian = le) @alignEnd(16)  {base usesEndian}
  block @padUntil(128) {
    r: u8  = records.length [auto]  input absent
    s: u8  = index.length [auto]  input absent
    recordsOfs: u64  = offsetof(records) [auto]  input absent
    indexOfs: u64  = offsetof(index) [auto]  input absent
    recPtrsOfs: u64  = offsetof(recPtrs) [auto]  input absent
  }
  pool: string<utf8>[until 0][until ""]
  records: Record($endian)[r]  at recordsOfs
  index: bytes[s]  at indexOfs
  recPtrs: *Record($endian) : u64 ref[r]  at recPtrsOfs
  layout {
    records[*].model @align(128)
    records[*].bank @align(128)
    records[*] @align(8)
    records @align(8)
    recPtrs
    index
  }
struct Record [Record] (endian endian: endian = le)  {usesEndian}
  model: *Model($endian) : u64
  bank: *Bank($endian) : u64
  curves: *Curves($endian) : u64
  motions: *Motion($endian) : u64
  layout {
    curves
    motions
  }
struct Model [Model] (endian endian: endian = le)  {usesEndian}
  m: u32
struct Bank [Bank] (endian endian: endian = le)  {usesEndian}
  e: u32
struct Curves [Curves] (endian endian: endian = le)  {usesEndian}
  x: *Curve($endian) : u64
  y: *Curve($endian) : u64
  layout {
    x.points
    y.points
    x
    y
    this
  }
struct Motion [Motion] (endian endian: endian = le)  {usesEndian}
  name: *string<utf8>[until 0] : u64 ref
struct Curve [Curve] (endian endian: endian = le)  {usesEndian}
  n: u64  = points.length [auto]  input absent
  ofs: u64  = offsetof(points) [auto]  input absent
  points: Pt($endian)[n]  at ofs
struct Pt [Pt] (endian endian: endian = le)  {usesEndian}
  a: f32
  b: f32
